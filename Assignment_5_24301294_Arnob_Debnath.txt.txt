import java.util.Scanner;
public class lab5{
  public static void main(String[]args){
  Scanner sc=new Scanner(System.in);
  while(true){
   System.out.println("Enter Number");
   int NUMBER=sc.nextInt();//TAKING NUMBER;
    int DIVISOR=0;//INITIAL DIVISOR;
    if(NUMBER%2==1 || NUMBER<0){//Condition for not taking negetive integer;
     break;
    }
    for(int AR=1;AR<=NUMBER;AR++){
      if(NUMBER%AR==0){
       DIVISOR++;
     
      }
    }
    System.out.println(NUMBER + " has " + DIVISOR+ " divisors");
  }
  }
}
