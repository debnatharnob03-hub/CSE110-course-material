    double AB=Math.sqrt(o);//Hypotenous;
    double SinA=BC/AB;
    double SinB=AC/AB;
    double CosA=AC/AB;
    double CosB=BC/AB;

    System.out.println("SinB="+SinB);
    System.out.println("CosA="+CosA);
    System.out.println("CosB="+CosB);

}

class lab2{
  public static void main(String[]args){
    double a=8;
    double b=3;//base
    double c=a/2;//height;
    double m=Math.pow(c,2);
    double n=Math.pow(b,2);
    double o=m+n;
    double d=Math.sqrt(o);
    System.out.println("Hypotenous="+d);
    double Circumfence=6*d;
    double p=Math.sqrt(3);
    double q=3*p/2;
    double s=d*d;
    double r=q*s;
    double Area_of_Hexagon=r;
    System.out.println("circumfence="+Circumfence);
    System.out.println("Area="+Area_of_Hexagon);
    
  }
}
