import java.util.Scanner;
class lab3{
  public static void main(String[]args){
    Scanner sc=new Scanner(System.in);
    int a=sc.nextInt();
    int b=sc.nextInt();
    String operator=sc.next();
    if(operator.equals("+")){
      System.out.println(a+b);
    }
    else{
      if(operator.equals("-")){
        System.out.println(a-b);
      }
      else{
        if(operator.equals("*")){
          System.out.println(a*b);
        }
        else{
          if(operator.equals("/")){
            System.out.println(a/b);
          }
        }
      }
    }
  }
}
